import re
from typing import Dict, List, Optional, Tuple

_WS_RE = re.compile(r"\s+")
_MACH_RE = re.compile(
    r"^\s*\[[^\]]+\]\s*(?:SAS:|CDISC:|ODM:|EDC:|Name=|Length=|DataType=)", re.I
)
_PLACEHOLDER_RE = re.compile(r"(?:_\s*){6,}")
_INT_RE = re.compile(r"^\s*\d{1,3}\s*$")
_LINE_NO_RE = re.compile(r"^\s*(\d{1,3}(?:\.\d+)?)\s*(?:\(\s*hidden\s*\))?\s*$", re.I)
_TIMEPOINTISH_RE = re.compile(r"[0-9]")
_EXCLUSION_RE = re.compile(r"^\s*Exclusion\s+\d+\.\s+\S", re.I)
_INCLUSION_RE = re.compile(r"^\s*Inclusion\s+\d+\.\s+\S", re.I)

_SHORT_COLON_MAXLEN = 40
_MID_X_MIN = 130.0
_MID_X_MAX = 420.0


def _clean(s: str) -> str:
    return _WS_RE.sub(" ", s or "").strip()


def _t(ln) -> str:
    return (getattr(ln, "text", "") or "").strip()


def _x0(ln) -> float:
    try:
        return float(getattr(ln, "x0", 0.0))
    except Exception:
        return 0.0


def _y0(ln) -> float:
    try:
        return float(getattr(ln, "y0", 0.0))
    except Exception:
        return 0.0


def _sz(ln) -> float:
    try:
        return float(getattr(ln, "size", 0.0))
    except Exception:
        return 0.0


def _is_bold(ln) -> bool:
    return bool(getattr(ln, "bold", False))


def _is_non_black(ln) -> bool:
    return bool(getattr(ln, "non_black", False))


def _is_machine_line(text: str) -> bool:
    t = (text or "").strip()
    if not t:
        return True
    if _MACH_RE.search(t):
        return True
    if t.startswith("[") and "]" in t[:60] and ("Name=" in t or "Length=" in t or "DataType=" in t):
        return True
    return False


def _group_rows(lines, y_tol: float = 2.5):
    if not lines:
        return []
    ls = sorted(lines, key=lambda z: (_y0(z), _x0(z)))
    rows = []
    cur = []
    cur_y = None
    for ln in ls:
        y = _y0(ln)
        if cur_y is None:
            cur = [ln]
            cur_y = y
            continue
        if abs(y - cur_y) <= y_tol:
            cur.append(ln)
        else:
            cur.sort(key=lambda z: _x0(z))
            rows.append(cur)
            cur = [ln]
            cur_y = y
    if cur:
        cur.sort(key=lambda z: _x0(z))
        rows.append(cur)
    return rows


def _row_text(row) -> str:
    parts = []
    for ln in row:
        t = _t(ln)
        if t:
            parts.append(t)
    return _clean(" ".join(parts))


def _row_has_entry_placeholder(row) -> bool:
    for ln in row:
        t = _t(ln)
        if not t or _is_machine_line(t):
            continue
        x = _x0(ln)
        if x > 420:
            continue
        if _PLACEHOLDER_RE.search(t) or "__" in t:
            return True
    return False


def _looks_like_option_text(t: str) -> bool:
    s = _clean(t).lower()
    if not s:
        return False
    if s in {"yes", "no", "na", "n/a", "unknown", "not applicable", "none"}:
        return True
    if len(s) <= 4 and s in {"y", "n"}:
        return True
    if re.match(r"^\(?\d+\)?$", s):
        return True
    if re.match(r"^[a-e]\.$", s) or re.match(r"^[a-e]\)$", s):
        return True
    if re.match(r"^(?:true|false)$", s):
        return True
    if s.startswith("□") or s.startswith("☐") or s.startswith("▢"):
        return len(s) <= 3
    if (s.startswith("o ") or s.startswith("O ".lower())) and len(s) <= 80:
        # common "radio" bullet rendered as letter O
        return True
    return False


def _is_left_colored_colon_label(ln) -> bool:
    t = _t(ln)
    if not t or not t.endswith(":"):
        return False
    if len(t) > _SHORT_COLON_MAXLEN:
        return False
    if _x0(ln) > 140:
        return False
    if not _is_bold(ln):
        return False
    if not _is_non_black(ln):
        return False
    sz = _sz(ln)
    return 8.0 <= sz <= 13.5


def _is_mid_colored_colon_label(ln) -> bool:
    t = _t(ln)
    if not t or not t.endswith(":"):
        return False
    if len(t) > _SHORT_COLON_MAXLEN:
        return False
    x = _x0(ln)
    if not (140 <= x <= 300):
        return False
    if not _is_bold(ln):
        return False
    if not _is_non_black(ln):
        return False
    sz = _sz(ln)
    return 8.0 <= sz <= 13.5


def _is_answers_label_row(row) -> bool:
    for ln in row:
        t = _t(ln)
        if not t:
            continue
        if t.rstrip() in {"Answer(s):", "Answers:", "Answer:"} and _is_mid_colored_colon_label(ln):
            return True
    for ln in row:
        t = _t(ln)
        if not t:
            continue
        if "answer" in t.lower() and _is_mid_colored_colon_label(ln):
            return True
    return False


def _row_right_line_number(row) -> Optional[float]:
    candidates = []
    for ln in row:
        t = _t(ln)
        if not t or _is_machine_line(t):
            continue
        x = _x0(ln)
        if x < 450:
            continue
        m = _LINE_NO_RE.match(t)
        if m:
            try:
                candidates.append(float(m.group(1)))
            except Exception:
                pass
    if len(candidates) == 1:
        return candidates[0]
    return None


def _row_left_timepoint_token(row) -> Optional[str]:
    best = None
    for ln in row:
        t = _t(ln)
        if not t or _is_machine_line(t):
            continue
        x = _x0(ln)
        if x > 120:
            continue
        s = _clean(t)
        if len(s) > 28:
            continue
        if not _TIMEPOINTISH_RE.search(s) and not (_PLACEHOLDER_RE.search(s) or "__" in s):
            continue
        if "?" in s:
            continue
        best = s
        break
    return best


def _bold_black_mid_candidates(row, min_size=8.5, max_size=13.5) -> List[str]:
    out = []
    for ln in row:
        t = _t(ln)
        if not t or _is_machine_line(t):
            continue
        x = _x0(ln)
        if not (_MID_X_MIN <= x <= _MID_X_MAX):
            continue
        if not _is_bold(ln):
            continue
        if _is_non_black(ln):
            continue
        sz = _sz(ln)
        if not (min_size <= sz <= max_size):
            continue
        s = _clean(t)
        if s:
            out.append(s)
    return out


def _black_mid_texts(row, allow_nonbold: bool = True, min_size=8.0, max_size=13.8) -> List[str]:
    out = []
    for ln in row:
        t = _t(ln)
        if not t or _is_machine_line(t):
            continue
        x = _x0(ln)
        if not (_MID_X_MIN <= x <= _MID_X_MAX):
            continue
        if _is_non_black(ln):
            continue
        if (not allow_nonbold) and (not _is_bold(ln)):
            continue
        sz = _sz(ln)
        if not (min_size <= sz <= max_size):
            continue
        s = _clean(t)
        if s:
            out.append(s)
    return out


def _activity_title_from_row(row) -> Optional[str]:
    if _row_left_timepoint_token(row) is None:
        return None
    if _row_right_line_number(row) is None:
        return None
    mids = _bold_black_mid_candidates(row, min_size=8.8, max_size=13.5)
    if not mids:
        return None

    for s in mids:
        if "?" in s:
            continue
        if s.endswith(":"):
            continue
        if len(s) < 4:
            continue
        if _EXCLUSION_RE.match(s) or _INCLUSION_RE.match(s):
            continue
        return _clean(s)
    return None


def _fallback_top_title_from_row(row) -> Optional[str]:
    mids = []
    for ln in row:
        t = _t(ln)
        if not t or _is_machine_line(t):
            continue
        if _is_non_black(ln):
            continue
        x = _x0(ln)
        if not (120 <= x <= 420):
            continue
        if not _is_bold(ln):
            continue
        sz = _sz(ln)
        if sz < 12.2:
            continue
        s = _clean(t)
        if not s or "?" in s or s.endswith(":"):
            continue
        mids.append((sz, s))
    if not mids:
        return None
    mids.sort(key=lambda z: (-z[0], len(z[1])))
    return mids[0][1]


def _schedule_name_from_rows(rows) -> Optional[str]:
    for row in rows:
        if not row:
            continue
        y = _y0(row[0])
        if y > 140:
            break
        left_label = None
        value = None
        for ln in row:
            t = _t(ln)
            if not t:
                continue
            x = _x0(ln)
            sz = _sz(ln)
            if x < 130 and _is_bold(ln) and (not _is_non_black(ln)) and 9.0 <= sz <= 13.5 and t.endswith(":"):
                left_label = t
            elif x >= 130 and (not _is_bold(ln)) and (not _is_non_black(ln)) and 9.0 <= sz <= 13.5:
                if value is None and len(t) >= 8 and (not _is_machine_line(t)):
                    value = t
        if left_label and value:
            return _clean(value)
    return None


def _extract_label_before_answers(rows, answers_ri: int, lower_bound_ri: int) -> str:
    collected_by_row: List[Tuple[int, List[str]]] = []
    started = False

    for ri in range(answers_ri - 1, lower_bound_ri - 1, -1):
        row = rows[ri]

        if _activity_title_from_row(row):
            break
        if _is_answers_label_row(row):
            break

        # Many layouts insert colored left labels (e.g., "Staff Initials:") between the
        # question label and "Answer(s):". Don't stop the upward scan until we've
        # actually started collecting the question label.
        if any(_is_left_colored_colon_label(ln) for ln in row):
            if started:
                break
            continue

        mids = _black_mid_texts(row, allow_nonbold=True)
        if not mids:
            if started:
                break
            continue

        non_option = [s for s in mids if not _looks_like_option_text(s)]
        if not non_option:
            if started:
                break
            continue

        started = True
        collected_by_row.append((ri, non_option))

        if any(s.endswith("?") for s in non_option):
            break
        if any(_EXCLUSION_RE.match(s) or _INCLUSION_RE.match(s) for s in non_option):
            break

    if not collected_by_row:
        return ""

    collected_by_row.sort(key=lambda z: z[0])
    parts: List[str] = []
    for _, segs in collected_by_row:
        parts.extend(segs)

    label = _clean(" ".join(parts))
    label = re.sub(r"\bAnswer(?:\(s\))?\s*:?\s*$", "", label, flags=re.I).strip()
    return label


def _extract_placeholder_label(rows, ri: int, lookback: int = 6) -> str:
    start = max(0, ri - lookback)
    parts: List[str] = []
    for k in range(start, min(len(rows), ri + 2)):
        row = rows[k]
        if _activity_title_from_row(row):
            continue
        mids_bold = _bold_black_mid_candidates(row, min_size=8.5, max_size=13.8)
        mids_any = _black_mid_texts(row, allow_nonbold=True)
        if mids_bold:
            for s in mids_bold:
                if not _looks_like_option_text(s):
                    parts.append(s)
        elif parts and mids_any:
            for s in mids_any:
                if not _looks_like_option_text(s):
                    parts.append(s)
    return _clean(" ".join(parts))


def extract(pages: List[Tuple[int, list]]) -> List[Dict[str, object]]:
    results: List[Dict[str, object]] = []

    current_form = ""
    current_schedule = ""

    pages_total = len(pages)
    pages_with_any = set()
    per_field_pages: Dict[str, set] = {}

    for page_index0, lines in pages:
        page_num = int(page_index0) + 1
        rows = _group_rows(lines)

        sched = _schedule_name_from_rows(rows)
        if sched:
            current_schedule = sched

        page_seen = set()

        last_activity_ri = -1
        last_answers_ri = -1

        for row in rows[:25]:
            if not row:
                continue
            if _y0(row[0]) > 210:
                break
            fb = _fallback_top_title_from_row(row)
            if fb:
                current_form = fb
                break

        ri = 0
        while ri < len(rows):
            row = rows[ri]

            act = _activity_title_from_row(row)
            if act:
                current_form = act
                last_activity_ri = ri
                last_answers_ri = ri
                ri += 1
                continue

            if _is_answers_label_row(row):
                form_name = current_form or current_schedule or ""
                lower = max(last_activity_ri + 1, last_answers_ri + 1, 0)
                label = _extract_label_before_answers(rows, ri, lower_bound_ri=lower)

                if label:
                    key = (form_name, label, page_num)
                    if key not in page_seen:
                        page_seen.add(key)
                        results.append({"form_name": form_name, "field_name": label, "page": page_num})
                        pages_with_any.add(page_num)
                        per_field_pages.setdefault(label, set()).add(page_num)

                last_answers_ri = ri
                ri += 1
                continue

            if _row_has_entry_placeholder(row) or (ri > 0 and _row_has_entry_placeholder(rows[ri - 1])):
                label = _extract_placeholder_label(rows, ri)
                if label:
                    form_name = current_form or current_schedule or ""
                    key = (form_name, label, page_num)
                    if key not in page_seen:
                        page_seen.add(key)
                        results.append({"form_name": form_name, "field_name": label, "page": page_num})
                        pages_with_any.add(page_num)
                        per_field_pages.setdefault(label, set()).add(page_num)

            mid_text = _row_text(row)
            if mid_text and (not _is_machine_line(mid_text)):
                if _EXCLUSION_RE.match(mid_text) or _INCLUSION_RE.match(mid_text):
                    form_name = current_form or current_schedule or ""
                    key = (form_name, mid_text, page_num)
                    if key not in page_seen:
                        page_seen.add(key)
                        results.append({"form_name": form_name, "field_name": mid_text, "page": page_num})
                        pages_with_any.add(page_num)
                        per_field_pages.setdefault(mid_text, set()).add(page_num)

            for ln in row:
                if _is_left_colored_colon_label(ln):
                    field_name = _clean(_t(ln))
                    if field_name:
                        form_name = current_form or current_schedule or ""
                        key = (form_name, field_name, page_num)
                        if key not in page_seen:
                            page_seen.add(key)
                            results.append({"form_name": form_name, "field_name": field_name, "page": page_num})
                            pages_with_any.add(page_num)
                            per_field_pages.setdefault(field_name, set()).add(page_num)

            ri += 1

    denom = max(1, pages_total)
    furniture = {"Staff Initials:", "Comment:"}
    filtered: List[Dict[str, object]] = []
    for rec in results:
        fn = rec.get("field_name", "") or ""
        fn = _clean(str(fn))
        if not fn:
            continue
        if fn in furniture:
            freq = len(per_field_pages.get(fn, set())) / denom
            if freq >= 0.70:
                continue
        if fn.endswith(":"):
            freq = len(per_field_pages.get(fn, set())) / denom
            if freq >= 0.90:
                continue
        filtered.append(rec)

    return filtered
