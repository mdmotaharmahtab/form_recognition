import re
import statistics
import unicodedata
from typing import List, Tuple, Dict, Any, Optional


_ID_RE = re.compile(r"^\[[A-Z0-9][A-Z0-9_]*\]$")
_BRACKET_RE = re.compile(r"^\[.*\]$")
_PAGE_RE = re.compile(r"^Page\s+\d+\s+of\s+\d+$", re.IGNORECASE)

_OPTION_WORDS = {
    "yes",
    "no",
    "normal",
    "abnormal",
    "not done",
    "na",
    "n/a",
    "not applicable",
    "unknown",
}


def _norm_space(s: str) -> str:
    return " ".join(s.split())


def _strip_leading_ordinals(s: str) -> str:
    s = s.strip()
    s = re.sub(r"^\s*\\\s*\d+\s*[.)]\s*", "", s)
    s = re.sub(r"^\s*\d+\s*[.)]\s*", "", s)
    return s.strip()


def _clean_label(s: str) -> str:
    s = s.replace("\u00a0", " ")
    s = _norm_space(s)
    s = s.strip()
    s = s.strip("*")
    s = s.strip()
    s = re.sub(r"^[\u2022•·]\s*", "", s)
    s = _strip_leading_ordinals(s)
    return s


def _text_quality(s: str) -> float:
    s = s.strip()
    if not s:
        return 0.0
    letters = 0
    nonspace = 0
    for ch in s:
        if ch.isspace():
            continue
        nonspace += 1
        if unicodedata.category(ch).startswith("L"):
            letters += 1
    if nonspace == 0:
        return 0.0
    return letters / nonspace


def _is_field_id_line(line) -> bool:
    t = (line.text or "").strip()
    return bool(_ID_RE.match(t))


def _is_bracket_meta(line) -> bool:
    t = (line.text or "").strip()
    return bool(_BRACKET_RE.match(t))


def _is_page_furniture(line) -> bool:
    t = _clean_label((line.text or "").strip())
    if not t:
        return True
    if _PAGE_RE.match(t):
        return True
    return False


def _median(vals: List[float], default: float) -> float:
    vals = [v for v in vals if v is not None]
    if not vals:
        return default
    try:
        return float(statistics.median(vals))
    except Exception:
        vals2 = sorted(vals)
        return float(vals2[len(vals2) // 2])


def _detect_body_size(lines) -> float:
    sizes = []
    for l in lines:
        if not getattr(l, "non_black", False) and (l.text or "").strip():
            if 6.0 <= float(getattr(l, "size", 0.0)) <= 12.5:
                sizes.append(float(l.size))
    return _median(sizes, 9.0)


def _detect_left_margin(lines, body_size: float) -> float:
    xs = []
    for l in lines:
        if not getattr(l, "non_black", False) and (l.text or "").strip():
            if float(l.size) <= body_size + 1.2 and 40.0 <= float(l.x0) <= 140.0:
                if 90.0 <= float(l.y0) <= 740.0:
                    xs.append(float(l.x0))
    return _median(xs, 64.0)


def _detect_form_title(lines, body_size: float) -> Optional[str]:
    cands = []
    for l in lines:
        if not getattr(l, "non_black", False):
            continue
        if float(l.x0) > 170.0:
            continue
        if float(l.y0) > 220.0:
            continue
        if float(l.size) < body_size + 4.0:
            continue
        t = (l.text or "").strip()
        if not t:
            continue
        if _is_bracket_meta(l):
            continue
        if _is_page_furniture(l):
            continue
        cands.append(l)

    if not cands:
        return None

    cands.sort(key=lambda z: (-float(z.size), float(z.y0), float(z.x0)))
    title = _clean_label(cands[0].text)
    return title or None


def _build_header_blocks(lines, body_size: float):
    header_lines = []
    for l in lines:
        if getattr(l, "non_black", False):
            continue
        if float(l.y0) > 175.0:
            continue
        if float(l.size) < body_size + 1.0:
            continue
        t = (l.text or "").strip()
        if not t:
            continue
        if _is_bracket_meta(l):
            continue
        if _is_page_furniture(l):
            continue
        header_lines.append(l)

    header_lines.sort(key=lambda z: (float(z.x0), float(z.y0)))

    blocks = []
    for l in header_lines:
        placed = False
        lx0, lx1 = float(l.x0), float(l.x1)
        for b in blocks:
            if abs(lx0 - b["x0_ref"]) <= 40.0 or not (lx1 < b["x0"] - 20.0 or lx0 > b["x1"] + 20.0):
                b["lines"].append(l)
                b["x0"] = min(b["x0"], lx0)
                b["x1"] = max(b["x1"], lx1)
                b["x0_ref"] = (b["x0_ref"] * 0.7) + (lx0 * 0.3)
                placed = True
                break
        if not placed:
            blocks.append({"x0": lx0, "x1": lx1, "x0_ref": lx0, "lines": [l]})

    out = []
    for b in blocks:
        b["lines"].sort(key=lambda z: (float(z.y0), float(z.x0)))
        txt = _clean_label(" ".join((ln.text or "").strip() for ln in b["lines"] if (ln.text or "").strip()))
        if not txt:
            continue
        out.append(
            {
                "x0": b["x0"],
                "x1": b["x1"],
                "x_center": 0.5 * (b["x0"] + b["x1"]),
                "text": txt,
            }
        )
    out.sort(key=lambda z: z["x_center"])
    return out


def _header_for_x(headers, x: float) -> Optional[str]:
    if not headers:
        return None
    best = None
    best_d = 1e18
    for h in headers:
        d = abs(float(h["x_center"]) - x)
        if d < best_d:
            best_d = d
            best = h
    if best is None:
        return None
    if best_d <= 130.0:
        return best["text"]
    return None


def _has_read_only_marker(lines, idx: int) -> bool:
    base = lines[idx]
    base_y = float(getattr(base, "y0", 0.0))
    base_x0 = float(getattr(base, "x0", 0.0))
    base_x1 = float(getattr(base, "x1", base_x0))
    base_xc = 0.5 * (base_x0 + base_x1)

    lo = max(0, idx - 8)
    hi = min(len(lines), idx + 14)
    for j in range(lo, hi):
        lj = lines[j]
        t = (lj.text or "").strip()
        if not t:
            continue
        tl = t.lower()
        if "read-only" not in tl and "read only" not in tl:
            continue

        y = float(getattr(lj, "y0", 0.0))
        if abs(y - base_y) > 75.0:
            continue

        lx0 = float(getattr(lj, "x0", 0.0))
        lx1 = float(getattr(lj, "x1", lx0))
        lxc = 0.5 * (lx0 + lx1)

        overlap = not (lx1 < base_x0 - 25.0 or lx0 > base_x1 + 25.0)
        near_center = abs(lxc - base_xc) <= 140.0
        if overlap or near_center:
            return True

    return False


def _find_inline_right_label(lines, id_idx: int, id_y: float, body_size: float) -> Optional[str]:
    id_line = lines[id_idx]
    id_x1 = float(getattr(id_line, "x1", 0.0))
    size_lo = body_size - 1.8
    size_hi = body_size + 2.0

    best = None
    best_key = None
    for j, lj in enumerate(lines):
        if j == id_idx:
            continue
        if getattr(lj, "non_black", False):
            continue
        if not (size_lo <= float(getattr(lj, "size", 0.0)) <= size_hi):
            continue
        y = float(getattr(lj, "y0", 0.0))
        if abs(y - id_y) > 10.0:
            continue
        if float(getattr(lj, "x0", 0.0)) <= id_x1 + 8.0:
            continue
        t = (lj.text or "").strip()
        if not t:
            continue
        if _is_bracket_meta(lj) or _is_page_furniture(lj):
            continue
        cand = _clean_label(t)
        if not cand:
            continue
        if cand.lower() in _OPTION_WORDS:
            continue
        dy = abs(y - id_y)
        dx = float(getattr(lj, "x0", 0.0)) - id_x1
        key = (dy, dx)
        if best_key is None or key < best_key:
            best_key = key
            best = cand

    if best and _text_quality(best) < 0.12 and len(best) <= 8:
        return None
    return best


def _find_left_label_block(lines, id_idx: int, id_y: float, body_size: float, left_margin: float, headers_present: bool) -> Optional[str]:
    y_window = 380.0
    size_lo = body_size - 1.8
    size_hi = body_size + 1.6
    x_hi = left_margin + 35.0

    best_j = None
    best_y = -1e18
    for j in range(id_idx - 1, -1, -1):
        lj = lines[j]
        y = float(lj.y0)
        if y < id_y - y_window:
            break
        if getattr(lj, "non_black", False):
            continue
        if headers_present and y <= 175.0 and float(lj.size) >= body_size + 1.0:
            continue
        if float(lj.x0) > x_hi:
            continue
        if not (size_lo <= float(lj.size) <= size_hi):
            continue
        t = (lj.text or "").strip()
        if not t:
            continue
        if _is_bracket_meta(lj) or _is_page_furniture(lj):
            continue
        if t in ("•", "\u2022"):
            continue
        ct = _clean_label(t)
        if ct.lower() in _OPTION_WORDS:
            continue
        if y > best_y:
            best_y = y
            best_j = j

    if best_j is None:
        return None

    start = best_j
    line_gap = max(18.0, body_size * 2.2)
    while start - 1 >= 0:
        prev = lines[start - 1]
        if getattr(prev, "non_black", False):
            break
        if float(prev.x0) > x_hi:
            break
        if not (size_lo <= float(prev.size) <= size_hi):
            break
        if (float(lines[start].y0) - float(prev.y0)) > line_gap:
            break
        pt = (prev.text or "").strip()
        if not pt:
            break
        if _is_bracket_meta(prev) or _is_page_furniture(prev):
            break
        if pt in ("•", "\u2022"):
            break
        pct = _clean_label(pt)
        if pct.lower() in _OPTION_WORDS:
            break
        start -= 1

    parts = []
    for k in range(start, best_j + 1):
        tk = (lines[k].text or "").strip()
        if not tk or tk in ("•", "\u2022"):
            continue
        if _is_bracket_meta(lines[k]) or _is_page_furniture(lines[k]):
            continue
        parts.append(tk)
    label = _clean_label(" ".join(parts))
    if not label:
        return None
    if len(label) <= 2:
        return None
    if label.lower() in _OPTION_WORDS:
        return None
    return label


def _find_left_label_block_below(lines, id_idx: int, id_y: float, body_size: float, left_margin: float, headers_present: bool) -> Optional[str]:
    y_window = 260.0
    size_lo = body_size - 1.8
    size_hi = body_size + 1.6
    x_hi = left_margin + 35.0

    best_j = None
    best_y = 1e18
    for j in range(id_idx + 1, len(lines)):
        lj = lines[j]
        y = float(lj.y0)
        if y > id_y + y_window:
            break
        if getattr(lj, "non_black", False):
            continue
        if headers_present and y <= 175.0 and float(lj.size) >= body_size + 1.0:
            continue
        if float(lj.x0) > x_hi:
            continue
        if not (size_lo <= float(lj.size) <= size_hi):
            continue
        t = (lj.text or "").strip()
        if not t:
            continue
        if _is_bracket_meta(lj) or _is_page_furniture(lj):
            continue
        ct = _clean_label(t)
        if not ct:
            continue
        if ct.lower() in _OPTION_WORDS:
            continue
        if y < best_y:
            best_y = y
            best_j = j

    if best_j is None:
        return None

    end = best_j
    line_gap = max(18.0, body_size * 2.2)
    while end + 1 < len(lines):
        nxt = lines[end + 1]
        if getattr(nxt, "non_black", False):
            break
        if float(nxt.x0) > x_hi:
            break
        if not (size_lo <= float(nxt.size) <= size_hi):
            break
        if (float(nxt.y0) - float(lines[end].y0)) > line_gap:
            break
        nt = (nxt.text or "").strip()
        if not nt:
            break
        if _is_bracket_meta(nxt) or _is_page_furniture(nxt):
            break
        nct = _clean_label(nt)
        if nct.lower() in _OPTION_WORDS:
            break
        end += 1

    parts = []
    for k in range(best_j, end + 1):
        tk = (lines[k].text or "").strip()
        if not tk:
            continue
        if _is_bracket_meta(lines[k]) or _is_page_furniture(lines[k]):
            continue
        parts.append(tk)
    label = _clean_label(" ".join(parts))
    if not label:
        return None
    if len(label) <= 2:
        return None
    if label.lower() in _OPTION_WORDS:
        return None
    return label


def _find_row_band_label(lines, id_y: float, body_size: float, left_margin: float) -> Optional[str]:
    size_lo = body_size - 2.0
    size_hi = body_size + 2.2
    x_hi = max(220.0, left_margin + 170.0)

    best = None
    best_key = None
    for lj in lines:
        y = float(getattr(lj, "y0", 0.0))
        if not (id_y - 70.0 <= y <= id_y + 12.0):
            continue
        if float(getattr(lj, "x0", 0.0)) > x_hi:
            continue
        if not (size_lo <= float(getattr(lj, "size", 0.0)) <= size_hi):
            continue
        t = (lj.text or "").strip()
        if not t:
            continue
        if _is_bracket_meta(lj) or _is_page_furniture(lj):
            continue
        ct = _clean_label(t)
        if not ct:
            continue
        if ct.lower() in _OPTION_WORDS:
            continue
        if ct in ("•", "\u2022"):
            continue

        dy = abs((id_y - 25.0) - y)
        x0 = float(getattr(lj, "x0", 0.0))
        key = (dy, x0, -len(ct))
        if best_key is None or key < best_key:
            best_key = key
            best = ct

    if best and len(best) <= 4 and _text_quality(best) < 0.45:
        return None
    return best


def _find_row_context(lines, id_idx: int, id_y: float, body_size: float, left_margin: float) -> Optional[str]:
    best = None
    best_key = None

    id_line = lines[id_idx]
    id_x = float(getattr(id_line, "x0", 0.0))

    for j in range(id_idx - 1, -1, -1):
        lj = lines[j]
        y = float(lj.y0)
        if y < id_y - 85.0:
            break
        t = (lj.text or "").strip()
        if not t:
            continue
        if _is_bracket_meta(lj) or _is_page_furniture(lj):
            continue
        if t in ("•", "\u2022"):
            continue
        if float(lj.size) >= body_size + 4.0 and float(lj.y0) <= 220.0:
            continue
        if not (id_y - 65.0 <= y <= id_y + 8.0):
            continue

        x0 = float(lj.x0)
        if x0 > 320.0 and abs((x0 + float(lj.x1)) * 0.5 - id_x) > 70.0:
            continue

        cand = _clean_label(t)
        if not cand:
            continue
        if cand.lower() in _OPTION_WORDS:
            continue

        dy = abs((id_y - 25.0) - y)
        key = (dy, x0)
        if best_key is None or key < best_key:
            best = cand
            best_key = key

    if best and len(best) <= 4 and _text_quality(best) < 0.45:
        return None
    return best


def _field_name_for_id(lines, id_idx: int, id_indices_by_y, headers, body_size: float, left_margin: float) -> Optional[str]:
    id_line = lines[id_idx]
    id_y = float(id_line.y0)
    id_xc = 0.5 * (float(id_line.x0) + float(id_line.x1))

    headers_present = bool(headers)
    col_header = _header_for_x(headers, id_xc) if headers_present else None

    multi_col = False
    y_bucket = int(id_y / 3.0)
    sibs = id_indices_by_y.get(y_bucket, [])
    if len(sibs) >= 2:
        xs = [float(lines[k].x0) for k in sibs]
        if max(xs) - min(xs) > 80.0:
            multi_col = True

    left_label = _find_left_label_block(lines, id_idx, id_y, body_size, left_margin, headers_present)
    if not left_label:
        left_label = _find_inline_right_label(lines, id_idx, id_y, body_size)

    row_label = None
    if headers_present and float(id_line.y0) > 175.0:
        row_label = _find_row_band_label(lines, id_y, body_size, left_margin)
    if not row_label:
        row_label = _find_row_context(lines, id_idx, id_y, body_size, left_margin)

    if not left_label and not row_label:
        left_label = _find_left_label_block_below(lines, id_idx, id_y, body_size, left_margin, headers_present)

    field = None
    if headers_present and float(id_line.y0) > 175.0:
        base = left_label or row_label
        if base and col_header and (multi_col or float(id_line.x0) >= 250.0):
            field = _clean_label(f"{base} - {col_header}")
        else:
            field = base or col_header
    else:
        field = left_label or row_label or col_header
        if field and col_header and multi_col and col_header not in field:
            field = _clean_label(f"{field} - {col_header}")

    if not field:
        return None

    q = _text_quality(field)
    if q < 0.12 and len(field) <= 8:
        return None
    if field.lower() in _OPTION_WORDS:
        return None
    if _PAGE_RE.match(field):
        return None
    return field


def extract(pages: List[Tuple[int, Any]]) -> List[Dict[str, Any]]:
    out: List[Dict[str, Any]] = []
    current_form = ""

    for page_idx0, lines in pages:
        if not lines:
            continue

        body_size = _detect_body_size(lines)
        left_margin = _detect_left_margin(lines, body_size)

        title = _detect_form_title(lines, body_size)
        if title:
            current_form = title

        headers = _build_header_blocks(lines, body_size)

        id_idxs = []
        for i, l in enumerate(lines):
            if _is_field_id_line(l):
                id_idxs.append(i)

        if not id_idxs:
            continue

        id_indices_by_y = {}
        for idx in id_idxs:
            yb = int(float(lines[idx].y0) / 3.0)
            id_indices_by_y.setdefault(yb, []).append(idx)

        for idx in id_idxs:
            if _has_read_only_marker(lines, idx):
                continue

            field_name = _field_name_for_id(lines, idx, id_indices_by_y, headers, body_size, left_margin)
            if not field_name:
                continue

            out.append(
                {
                    "form_name": current_form or "",
                    "field_name": field_name,
                    "page": int(page_idx0) + 1,
                }
            )

    return out
